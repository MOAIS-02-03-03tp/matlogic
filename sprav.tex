    \subsection{Основные операции}
    \begin{itemize}
        \item $a \subset A$
        \item $\varnothing$
        \item $A \subset B \Leftrightarrow A \subseteq B \text{ и } \exists x (x \in B \text{ и } x\not \int A)$
        \item $A \subseteq B \Leftrightarrow \forall x (x \in A \Rightarrow x \in B)$
        \item $A \land B = \{x | x \in A \text{ и } x \in B\}$
        \item $A \lor B = \{x | x \in A \text{ или } x \in B\}$
        \item $A \ B = \{x | x\in A \text{ и } x \not \in B\}$
        \item $A \bigtriangleup B = (A \lor B)\backslash(A\land B) = (A\backslash B)\lor(B \backslash A)$
        \item $\overline{A_B} = B \backslash A = \{x \in B | x \not in A\}$
    \end{itemize}
    $A_1$x$\dots$x$A_n$=$\{(a_1,\dots,a_n) | \forall i \in n (a_i \in A_i)\}$, где ($a_1,\dots,a_n$) - упорядоченный набор, который определяется
    следующим образом:
    \begin{enumerate}
        \item $n=0 \Rightarrow \varnothing$
        \item $n=1 \Rightarrow \{a_1\}$
        \item $n=2 \Rightarrow \{\{a_1\},\{a_2\}\}(\{\{a_2\},\{a_2,a_1\}\})$
        \item $n > 2 \Rightarrow (a_1,\dots,a_{n-1},a_n)=((a_1,\dots,a_{n-1}),a_n)$
    \end{enumerate}
    \[(1,2,3)=((1,2),3)=\{\{(1,2)\},\{3\}\}=\{\{1\},\{2\},\{3\}\}=\{\{\{1\},\{1,2\}\}\},\{\{\{1\},\{1,2\}\},3\}\]
    \subsection{Бинарные отношения}
    \underline{Определение: } Множество $\rho \supseteq A_1 \text{x}\dots \text{x}A_n$ - называется n-местным отношением на ($A_1,\dots,A_n$)
    (n>0). Если $A_1 =A_2 = \dots = A_n = A$, то $\rho$ называется n-местным отношением на A. Если n=2, то $\rho$ называется бинарным отношением.
    \subsection*{Пример:}
    \[A=\{1,2,3\}\]
    \[B=\{a,b\}\]
    \[A \text{ x } B = \{(1;a),(2;a),(3;a),(1,b),(2,b),(3,b)\}\]
    \[B \text{ x } A = \{(a;1),(a;2),(a;3),(b,1),(b,2),(b,3)\}\]
    \begin{itemize}
        \item $\rho_1,\rho_2 \subseteq A \text{ x } B$
        \item $\rho_1 = \{(1;a),(1;b)\}$
        \item $\rho_2 = \{(1;a),(1;b),(2;b)\}$
        \item $\rho_3 \subseteq A^2$
        \item $\rho_3 = \{(1,1),(1,2),(3;1)\}$
        \item $\rho_4,\rho_5 \subseteq A \text{ x } B \text{ x } A$
        \item $\rho_4 = \{(1,b,2),(3,a,1)\}$
        \item $\rho_5 = \varnothing$
    \end{itemize}
    \underline{Определение: } Пусть $\rho$ = A x B
    \[\text{Dom}(\rho)\overset{\text{def}}{=}\{x \in A | \exists y \in B:(x,y) \in \rho\} \text{ - область определения}\]
    \[\text{Im}(\rho)\overset{\text{def}}{=}\{y \in B | \exists x \in A:(x,y)\in \rho\} \text{ - область значений}\]
    \[\rho^{-1}\overset{\text{def}}{=} \{(y,x)\in B \text{ x } A | (x,y)\in \rho\} \text{ - обратное к } \rho\]
    \underline{Пример:} 
    \[\text{Dom}(\rho_2)=\{1,2\}\]
    \[\text{Im}(\rho_2)=\{a,b\}\]
    \[\rho^{-1}=\{(a,1),(b,1),(b,2)\}\]
    \subsection*{Своиства бинарный отношений на A:}
    Пусть $\rho \subseteq A^2$. Тогда:
    \begin{enumerate}
        \item $\rho$ - рефлексивное $\overset{\text{def}}{\Leftrightarrow} \forall x \in A ((x,x)\in \rho)$
        \item $\rho$ - симметричное $\overset{\text{def}}{\Leftrightarrow} \forall x \in A ((x,y)\in \rho \Rightarrow (y,x) \in \rho)$
        \item $\rho$ - транзитивно $\overset{\text{def}}{\Leftrightarrow} \forall x,y,z \in A ((x,y)\in \rho \text{ и } (y,z)\in \rho
        \Rightarrow (x,z) \in \rho)$
        \item $\rho$ - антирефлексивно $\overset{\text{def}}{\Leftrightarrow} \forall x \in A ((x,x) \not \in \rho)$
        \item $\rho$ - антисимметрично $\overset{\text{def}}{\Leftrightarrow} \forall x \in A (x \not = y \text{ и } (x,y) \in \rho
        \Rightarrow (y,x)\not \in \rho)$
        \item $\rho$ - связное $\overset{\text{def}}{\Leftrightarrow}\forall x,y \in A (x \not = y \Rightarrow (x,y)\in \rho \text{ или }
        (y,x)\in \rho)$
    \end{enumerate}

    $\frac{a}{\rho}$=$\{x \in A | <a,x> \in \rho\}$ называется классов эквивалентности с представителем
    \subsubsection*{Теорема 1}\label{th:1}
    \par\noindent
    Пусть $\rho$ - эквивалентность на А, a и b $\in A$. Тогда $<a,b> \in \rho \Leftrightarrow \frac{a}{\rho}=\frac{b}{\rho}.$\\
    \par\noindent
    \underline{Доказательство:}
    \begin{adjustwidth}{1.5em}{1.5em}
        $\boxed{\Rightarrow}$ Пусть $x \in \frac{a}{\rho}.$ По определению $<a,x> \in \overset{(1)}{\rho}$. \\
        Так как $<a,b>\in \rho$ и $\rho$ - симметрично, то $<b,a> \in \overset{(2)}{\rho}$\\
        (1)(2) $\underset{\rho \text{ - транзитивно}}{\Rightarrow} <b,x> \in \rho \Rightarrow x \in \frac{b}{\rho}$.\\
        То $\frac{a}{\rho} \subseteq \frac{b}{\rho}.$ Аналогично, $\frac{b}{\rho} \subseteq \frac{a}{\rho}$. 
        Следовательно $\frac{a}{\rho}=\frac{b}{\rho}$\\
        $\boxed{\Leftarrow}$ Так как - $\rho$ рефлексивно, то $<a,a> \in \rho,$ то есть $a \in \frac{a}{\rho}$.\\
        Из $\frac{a}{\rho}=\frac{b}{\rho}$ следует, что $a \in \frac{b}{\rho}.$ Поэтому $<b,a> \in \rho$, и за счёт
        метричности $\rho$ получим, что $<a,b> \in \rho$
    \end{adjustwidth}

    \subsection{Отношение порядка}
    \underline{Определение:} Бинарное отношение $\rho \subseteq A^2$ называется отношением порядка, если
    оно антисимметрично и транзитивно.
    \begin{itemize}
        \item Если $\rho$ \underline{рефлексивно}, то оно называется \underline{нестрогим} порядком.
        \item Если $\rho$ \underline{антирефлексивно} - \underline{строгим} порядком.
        \item Если $\rho$ связно - \underline{линейным} порядком
        \item Если на $A$ введено отношение порядка $\rho$, то $A$ называют упорядоченным с помощью
        $\rho$ множеством.
    \end{itemize}
    \underline{Пример:}:
    \begin{enumerate}
        \item[\circled{1}] $\leq$ на $Z$\\
        \[<x,y> \in \leq \Leftrightarrow y-x \geq 0 \Leftrightarrow (\exists z \in N: x+z=y) 
        \text{ или } x=y\]

        $\hspace{10pt}\begin{matrix}
            1) &
            \begin{matrix}
                x\leq y\\  
                y \leq x
            \end{matrix} &
            \Big\} &\Rightarrow x=y & \text{ антисимметрично}\\
            2) &
            \begin{matrix}
                x \leq y\\  
                y \leq z
            \end{matrix} &
            \Big\} & x \leq z & \text{ транзитивно}
        \end{matrix}
        \Bigg\} \leq - \text{ порядок}$
        \begin{enumerate}
            \item[3)] $x\leq x - \text{рефлексивно} \Rightarrow \leq \text{нестрогий порядок}$
            \item[4)] $x \not = y \Rightarrow x \leq y \text{ или } y \leq x \text{ связно }
            \Rightarrow \leq - \text{ линейный порядок}$ 
        \end{enumerate}
        \item[\circled{2}] $\subseteq \text{ на } P(A), \text{ где } A = \{a,b\}$\\
        \[<x,y> \in \subseteq \Leftrightarrow x \subseteq, \text{ где } x,y \in A\]
        \begin{enumerate}
            \item[1)] 
            $\begin{matrix}
                x \subseteq y\\
                y \subseteq x
            \end{matrix} \Bigg\} \Rightarrow x=y$
            \item[2)] 
            $\begin{matrix}
                x \subseteq y\\
                y \subseteq z
            \end{matrix} \Bigg\} \Rightarrow x \subseteq z$
            \item[3)] $x \subseteq x$\\
            \[1),2),3) \Rightarrow \subseteq - \text{ нестрогий порядок}\]
            \item[4)] $\{a\} \not = \{b\}, \text{ но } \{a\}\not \subseteq \{b\} 
            \text{ и } \{b\} \not \subseteq \{a\}$
            \[\subseteq - \text{ нелинейный порядок}\]
        \end{enumerate}
    \end{enumerate}
    \pagebreak
    \underline{Определение: } Пусть $\rho$ - порядок на $A$, $a \in A, B \subseteq  A$.
    \begin{enumerate}
        \item a - \underline{минимальный} $\overset{\text{def}}{\Leftrightarrow} \forall x \in A
        (<x,a> \in \rho \Rightarrow x=a)$
        \item a - \underline{максимальный} $\overset{\text{def}}{\Leftrightarrow} \forall x \in A
        (<a,x> \in \rho \Rightarrow x=a)$
        \item a - \underline{наименьший} $\overset{\text{def}}{\Leftrightarrow} \forall x \in A
        (a \not = x \Rightarrow <a,x> \in \rho)$
        \item a - \underline{наибольший} $\overset{\text{def}}{\Leftrightarrow} \forall x \in A
        (a \not = x \Rightarrow <x,a> \in \rho)$
        \item a - \underline{нижняя грань} множества B, если $\forall x \in B
        (<a,x> \in \rho \text{ или } a=x)$
        \item a - \underline{верхняя грань} множества B, если $\forall  x \in B 
        (<x,a> \in \rho \text{ или } x=a)$
        \item Наибольшая/наименьшая из всех нижних/верхних граней( если она $\exists$ ) \\
        называется
        \underline{нижней/верхней границей} множества $B$.
    \end{enumerate}
    \underline{Пример:} $A = N \textbackslash \{1\}$
    \[<x,y> \in \rho \Leftrightarrow x | y, \text{ т.е. } x - \text{ делитель }y.\]
    $B=\{4,7,12\}$
    \begin{itemize}
        \item $\forall x \in A ( x| 2 \Rightarrow x=2)$ минимальные $2,3,5,7,\dots$ - простые
        \item $\forall x \in A ( x| 3 \Rightarrow x=3)$ максимальные \textemdash 
        \item $a$ - максимальное $\Leftrightarrow \forall x \in A (a|x \Rightarrow x=a)$
        \item $a$ - наименьшее $\Leftrightarrow \forall x \in A (a \not = x \Rightarrow a|x)$ 
        наименьшее \textemdash , наибольшее \textemdash
        \item $a$ - нижняя грань $B \Leftrightarrow \forall x \in B (a|x \text{ или } x=a)$
        \item $a$ - верхняя грань $B: 84,168,252,\dots \vdots 84$
    \end{itemize}
    \subsection{Отображения}
    \underline{Определение: } Бинарное отношение $\varphi \subseteq A x B$ называется отображением из $A$ в $B$,
    если 
    \begin{enumerate}
        \item $\forall x \in A \exists y \in B: <x,y> \in \varphi$
        \item $\forall x \in A \forall y_1,y_2 \in B 
        (<x,y_1> \in \varphi \text{ и } <x,y_2> \in \varphi \Rightarrow y_1 = y_2)$
    \end{enumerate}
    \underline{Обозначение: } $<x,y> \in \varphi$ заменим на $\varphi(x) = y$
    \pagebreak
    
    \underline{Определение: } Пусть: $A \rightarrow B$
    \begin{enumerate}
        \item $\varphi$ инъективно $\Leftrightarrow \forall x_1,x_2 \in A (x_1 \not = x_2 
        \Rightarrow \varphi(x_1) \not = \varphi(x_2) )$
        \item  $\varphi$ сюръективно $\Leftrightarrow \forall y \in B \exists x \in A : \varphi(x) = y$
        \item $\varphi$ биективно $\Leftrightarrow \varphi$ инъективно и сюръективно
        \item Если $\varphi^{-1}$(бинарное отношение, обратное к $\varphi$) является тоже отображением на $BxA$,
        то $\varphi^{-1}$ называется \underline{обратным к $\varphi$ отображением}.
    \end{enumerate}

    \subsubsection*{Теорема 3}\label{th:3}
    \par\noindent
    Если $\varphi$ и $\Psi$ инъективны/сюръективны/биективны, то $\varphi \Psi$ инъективно/сюръективно/биективно.\\
    \underline{Доказательство:}
    \begin{adjustwidth}{1.5em}{1.5em}
        \begin{enumerate}
            \item Пусть $\varphi: A \rightarrow B$ и $\Psi: B \rightarrow C$ инъективно.\\
            Тогда $\Psi(\varphi(a_1))=\Psi(\varphi(a_2))\overset{\Psi \text{ин.}}{\Rightarrow} a_1=a_2.$
            Т.е. $\varphi \Psi$ - инъективно.
            
            \item Пусть $\varphi: A \rightarrow B$ и $\Psi: B \rightarrow C$ сюръективны, $c \in C.$\\
            Т.к. $\Psi$ - сюръективно $\Rightarrow \exists b \in B: \Psi(b)=C.$ \\
            Т.к. $\varphi$ - сюръективно $\Rightarrow \exists a \in A:\varphi(a)=b$.\\
            Следовательно $\intercal c \in C \exists a \in A:\Psi(\varphi(a))=\Psi(b)=C,$т.е. $\varphi \Psi$ сюръективно.
        
            \item Т.к. $\varphi \Psi$ сюръективно и инъективно $\overset{\text{опр.}}{\Rightarrow}$ биективно.
        \end{enumerate}
